function draw() {
   background("lightblue");
   for (let x = 0; x < width; x += 10) {
	 stroke(x, 0, 0);
       line(x, 0, x, height)
   };
   for (let y = 0; y < height; y += 10) {
	stroke(0, y, 0);
       line(0, y, width, y)
   };
}
